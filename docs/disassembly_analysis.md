# Дизассемблирование

asm
movl $0, -4(%rbp)      # sum = 0
movl $1, -8(%rbp)      # i = 1
jmp .L2                # переход к проверке условия
.L3:
movl -8(%rbp), %eax    # eax = i
addl %eax, -4(%rbp)    # sum += i
addl $1, -8(%rbp)      # i += 1
.L2:
cmpl $10, -8(%rbp)     # сравнить i с 10
jle .L3                # если i <= 10 -> тело цикла
movl -4(%rbp), %eax    # результат (sum) в eax -> возвращаемое значение


- -4(%rbp) — локальная переменная-аккумулятор (sum), начало 0.
- -8(%rbp) — счётчик цикла (i), начало 1.
- jmp .L2 переходит сразу к проверке (цикл с предусловием).
- jle — «перейти, если меньше или равно», т.е. цикл идёт, пока i <= 10.
- Итог: сумма чисел от 1 до 10 = 55.

## Эквивалент на Python

python
total = 0
i = 1
while i <= 10:
    total += i
    i += 1
print(total)  # 55
